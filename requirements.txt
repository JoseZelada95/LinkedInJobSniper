# ── Dependencias de ejecución del agente ────────────────────────────────────
# Instalación:  pip install -r requirements.txt
# Ordenadas por bloque funcional para que sea fácil ver para qué sirve cada una.

# ⚠️  Las versiones usan rangos amplios (>=). Para un entorno reproducible,

# Scraping de ofertas
python-jobspy            # buscador multi-sitio (LinkedIn, Indeed, Glassdoor)
pandas                   # tratamiento tabular de las ofertas

requests                 # descarga HTTP directa
fake_useragent           # cabeceras de navegador aleatorias (evita bloqueos)

# Configuración y modelos de datos
python-dotenv            # lectura del fichero .env
pydantic                 # validación de la salida estructurada del LLM

# LLM (LangChain + cliente compatible con OpenAI → Groq por defecto)
langchain>=0.1.0

openai                   # cliente base que usa langchain-openai

pypdf
pymupdf                  # PyMuPDF: generar y leer los PDFs personalizados

# Google (Sheets de seguimiento + Drive para el CV)
google-api-python-client
google-auth
google-auth-httplib2
google-auth-oauthlib

# Utilidades varias
markdown2                # usado por dependencias de jobspy para descripciones
